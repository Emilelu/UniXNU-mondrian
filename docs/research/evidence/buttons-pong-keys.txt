static void look(struct key *k,bool down)
{
    bool steady=k->last==down;k->last=down;
    if (!k->usable || !steady) return;
    if (!k->live) {if (!down) k->live=true;return;}
    if (k->down==down) return;
    k->down=down;
    pong_touch_post(down?PONG_BUTTON_DOWN:PONG_BUTTON_UP,k->code,0,0);
    printf("Pong keys: %u %s\n",k->code,down?"down":"up");
}

static void keys_thread(__unused void *arg,__unused wait_result_t result)
{
    if (!configuration()) {printf("Pong keys: missing/incompatible DT; SKIP\n");goto out;}
    uint32_t version=rd(core);
    if (version<0x50000000 || version>=0x80000000) {
        printf("Pong keys: unsupported arbiter version %x; SKIP\n",version);goto out;
    }
    v7=version>=0x70000000;channels=rd(core+4)&0x7ff;
    if (!channels || channels>(v7?1024u:512u)) {
        printf("Pong keys: invalid channel count %u; SKIP\n",channels);goto out;
    }
    find_channel(&pon);find_channel(&pin);
    uint8_t id[2],state=0,c[8];
    if (read_registers(&pon,4,id,2) && id[0]==1 && id[1]==9 &&
        read_registers(&pon,0x10,&state,1)) {
        keys[0].usable=keys[2].usable=true;
        keys[0].last=(state&0x80)!=0;keys[2].last=(state&0x40)!=0;
    }
    if (read_registers(&pin,4,id,2) && id[0]==0x10 && id[1]>=0x10 && id[1]<=0x13 &&
        read_registers(&pin,0x40,c,8) && (c[0]&3)==0 && (c[2]&7)<4 && (c[6]&0x80) &&
        read_registers(&pin,0x10,&state,1)) {
        keys[1].usable=true;keys[1].last=(state&1)==0;
    }
    if (!(keys[0].usable || keys[1].usable) || !pong_buttons_device_ready()) {
        printf("Pong keys: no verified input peripheral; SKIP\n");goto out;
    }
    for (unsigned i=0;i<3;i++) keys[i].live=keys[i].usable && !keys[i].last;
    printf("Pong keys: arbiter v%u channels=%u PON=%u GPIO6=%u; power=%u up=%u down=%u\n",
        v7?7:5,channels,pon.channel,pin.channel,keys[0].usable,keys[1].usable,keys[2].usable);
    unsigned failures=0;
    for (;;) {
        bool good=true;
        if (keys[0].usable) {
            if (read_registers(&pon,0x10,&state,1)) {
                look(&keys[0],(state&0x80)!=0);look(&keys[2],(state&0x40)!=0);
            } else good=false;
        }
        if (keys[1].usable) {
            if (read_registers(&pin,0x10,&state,1)) look(&keys[1],(state&1)==0);
            else good=false;
        }
        failures=good?0:failures+1;
        if (failures>=8) {
            for (unsigned i=0;i<3;i++) if (keys[i].down)
                pong_touch_post(PONG_BUTTON_UP,keys[i].code,0,0);
            printf("Pong keys: repeated observer read failure; stop optional polling\n");break;
        }
        uint64_t deadline;clock_interval_to_deadline(16,NSEC_PER_MSEC,&deadline);
        assert_wait_deadline((event_t)keys,THREAD_UNINT,deadline);thread_block(THREAD_CONTINUE_NULL);
    }
out:
    thread_terminate(current_thread());
}

void pong_buttons_init(void)
{
    uint32_t enabled=0;thread_t t;
    if (!PE_parse_boot_argn("pong_buttons",&enabled,sizeof(enabled)) || !enabled) return;
    if (kernel_thread_start(keys_thread,NULL,&t)==KERN_SUCCESS) thread_deallocate(t);
}
